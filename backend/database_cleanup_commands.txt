# PostgreSQL Database Cleanup Commands
# Run these commands one by one in PowerShell

# Step 1: Check existing database
psql -U postgres -c "\l" | findstr opticare

# Step 2: Drop existing database (if exists)
psql -U postgres -c "DROP DATABASE IF EXISTS opticare_db;"

# Step 3: Drop existing user (if exists) 
psql -U postgres -c "DROP USER IF EXISTS opticare_user;"

# Step 4: Create fresh database
psql -U postgres -c "CREATE DATABASE opticare_db;"

# Step 5: Create dedicated user
psql -U postgres -c "CREATE USER opticare_user WITH PASSWORD 'opticare2024!';"

# Step 6: Grant permissions
psql -U postgres -c "GRANT ALL PRIVILEGES ON DATABASE opticare_db TO opticare_user;"

# Step 7: Grant schema permissions (PostgreSQL 15+)
psql -U postgres -d opticare_db -c "GRANT ALL ON SCHEMA public TO opticare_user;"

# Step 8: Verify setup
psql -U postgres -c "\l" | findstr opticare
psql -U postgres -c "\du" | findstr opticare
